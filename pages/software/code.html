04 — Software

Arduino Firmware

Core sketch (`AquaSense_Phase1.ino`) running on the Arduino Uno microcontroller.

code/AquaSense_Phase1.ino C++ / Arduino 
/*
  Project: AquaSense Phase 1 Prototype
  Controller: Arduino Uno
  Components: HC-SR04 Ultrasonic Sensor, SG90 Servo Motor, 16x2 LCD
*/

#include <Servo.h>
#include <LiquidCrystal.h>

const int trigPin = 9;
const int echoPin = 10;
const int servoPin = 6;

Servo waterServo;
LiquidCrystal lcd(12, 11, 5, 4, 3, 2);

void setup() {
  pinMode(trigPin, OUTPUT);
  pinMode(echoPin, INPUT);
  waterServo.attach(servoPin);
  waterServo.write(0); // Initial closed state
  
  lcd.begin(16, 2);
  lcd.print("AquaSense Ready");
}

void loop() {
  long duration;
  int distance;

  digitalWrite(trigPin, LOW);
  delayMicroseconds(2);
  digitalWrite(trigPin, HIGH);
  delayMicroseconds(10);
  digitalWrite(trigPin, LOW);

  duration = pulseIn(echoPin, HIGH);
  distance = duration * 0.034 / 2;

  if (distance > 0 && distance < 15) {
    lcd.clear();
    lcd.print("Dispensing...");
    waterServo.write(90); // Open valve
    delay(3000);
    waterServo.write(0);  // Close valve
  } else {
    lcd.setCursor(0, 0);
    lcd.print("Place Glass    ");
  }
  delay(500);
}
